# React (если это Vite-проект)
npm install react react-dom

# Роутинг (для вкладок "Магазин", "Наковальня", "Эпоха Огня")
npm install react-router-dom

# Стейт-менеджер (Глобальное состояние инвентаря и валюты)
npm install zustand

# Анимации ("Сок", пульсации, красивые переходы)
npm install framer-motion

# Эффект конфетти (для успешного крафта в наковальне)
npm install canvas-confetti

# Иконки (опционально, но обычно используются для UI, например Lucide)
npm install lucide-react

# Vite и плагины
npm install -D vite @vitejs/plugin-react typescript

# Tailwind CSS v4 (в 4-й версии используется Vite-плагин)
npm install -D tailwindcss @tailwindcss/vite

# Типы для TypeScript
npm install -D @types/react @types/react-dom @types/node @types/canvas-confetti
